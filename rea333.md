    @property
    def postgres_dsn(self) -> str:
        hosts: list[str] = []
        for raw_host in self.postgres_hosts.split(","):
            address = raw_host.strip()
            host, separator, port_text = address.rpartition(":")
            if not separator or not host or not port_text.isdigit():
                raise ValueError("POSTGRES_HOSTS: каждый адрес должен иметь вид host:port")
            port = int(port_text)
            if port < 1 or port > 65535:
                raise ValueError("POSTGRES_HOSTS: порт должен быть от 1 до 65535")
            hosts.append(f"{host}:{port}")
        if len(hosts) not in (1, 2) or len(set(hosts)) != len(hosts):
            raise ValueError("POSTGRES_HOSTS: нужен один или два разных адреса")
        url = URL.create(
            "postgresql+asyncpg",
            username=self.db_user,
            password=self.db_password,
            database=self.db_name,
            query={"host": tuple(hosts), "target_session_attrs": "read-write"},
        )
        return url.render_as_string(hide_password=False)

    def analyzers_for(self, manager_id: str) -> tuple[str, ...] | None:
        """Возвращает обязательную цепочку анализаторов или отказывает во входе."""

        """Разрешает задавать set как JSON-массив или строку через запятую."""
        if value is None or value == "":
            return set()
        if isinstance(value, str):
            items = (
                json.loads(value)
                if value.lstrip().startswith("[")
                else value.split(",")
            )
            result = set()
            for item in items:
                manager_id = str(item).strip()
                if manager_id:
                    result.add(manager_id.zfill(8))
            return result
        return value

    def validate_sampling_parameters(self) -> "Settings":
        configured = (
            self.prompt_analyzer_manager_ids
            | self.audio_analyzer_manager_ids
            | self.factor_analyzer_manager_ids
        )
        unknown = configured - self.analyzed_manager_ids
        without_profile = self.analyzed_manager_ids - configured
        without_prompt = self.analyzed_manager_ids - self.prompt_analyzer_manager_ids
        if unknown:
            raise ValueError(
                f"Профиль задан для менеджеров вне ANALYZED_MANAGER_IDS: {sorted(unknown)}"
            )
        if without_profile:
            raise ValueError(
                f"Для менеджеров не назначен ни один анализатор: {sorted(without_profile)}"
            )
        if without_prompt:
            raise ValueError(
                "Prompt Analyzer с гигиеной, вкладами и рисками обязателен для "
                f"менеджеров: {sorted(without_prompt)}"
            )
        if self.audio_analyzer_manager_ids and not self.audio_analyzer_url:
            raise ValueError("Для Audio Analyzer требуется AUDIO_ANALYZER_URL")
        if self.factor_analyzer_manager_ids and not self.factor_analyzer_url:
            raise ValueError("Для Factor Analyzer требуется FACTOR_ANALYZER_URL")

        role_sets = {
            "EFS_VIP_KM": self.efs_vip_km_manager_ids,
            "EFS_VIP_AKM": self.efs_vip_akm_manager_ids,
            "EFS_VIP_DSB": self.efs_vip_dsb_manager_ids,
            "EFS_VIP_TFS": self.efs_vip_tfs_manager_ids,
        }
        seen: dict[str, str] = {}
        for role, manager_ids in role_sets.items():
            for manager_id in manager_ids:
                previous_role = seen.get(manager_id)
                if previous_role:
                    raise ValueError(f"Manager {manager_id} in role {previous_role} and {role}")
                seen[manager_id] = role
        missing_roles = self.analyzed_manager_ids - set(seen)
        if missing_roles:
            raise ValueError(f"every analyzing manager should have role: {missing_roles}")

        return self

    pool_size=settings.postgres_pool_size,

    },

    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,

def get_database_engine() -> AsyncEngine:
    return engine


def is_failover_error(exc: DBAPIError) -> bool:
    if exc.connection_invalidated:
        return True
    original = exc.orig
    sqlstate = getattr(original, "sqlstate", None) or getattr(original, "pgcode", None)
    if sqlstate is None:
        cause = getattr(original, "__cause__", None)
        sqlstate = getattr(cause, "sqlstate", None) or getattr(cause, "pgcode", None)
    if not isinstance(sqlstate, str):
        return False
    return sqlstate.startswith("08") or sqlstate in {
        "25006",  # read_only_sql_transaction
        "57P01",  # admin_shutdown
        "57P02",  # crash_shutdown
        "57P03",  # cannot_connect_now
    }


@asynccontextmanager
async def session_scope() -> AsyncIterator[AsyncSession]:
    try:
        async with session_factory() as session, session.begin():
            yield session
    except DBAPIError as exc:
        if is_failover_error(exc):
            await engine.dispose()
        raise


async def wait_for_database() -> None:
    attempt = 0
    while True:
        attempt += 1
        try:
            async with engine.connect() as connection:
                await connection.execute(text("SELECT 1"))
            logger.info("PostgreSQL доступен после попытки %d", attempt)
            return
        except (SQLAlchemyError, OSError, TimeoutError):
            if attempt == 1 or attempt % 12 == 0:
                logger.warning(
                    "PostgreSQL пока недоступен; попытка %d, повтор через 5 секунд",
                    attempt,
                )
            await asyncio.sleep(5)


async def check_database() -> bool:
    try:
        async with engine.connect() as connection:
            await connection.execute(text("SELECT 1"))
    except (SQLAlchemyError, OSError, TimeoutError):
        return False
    return True


async def initialize_database() -> None:
    if not settings.database_auto_create_schema:
        return
    async with engine.begin() as connection:
        await connection.execute(text(f'CREATE SCHEMA IF NOT EXISTS "{SCHEMA}"'))
        await connection.run_sync(Base.metadata.create_all)

target_metadata = Base.metadata
sync_url = make_url(get_settings().postgres_dsn).set(
    drivername="postgresql+psycopg2"
)


def is_temporary_database_error(exc: DBAPIError) -> bool:
    if exc.connection_invalidated:
        return True
    original = exc.orig
    sqlstate = getattr(original, "pgcode", None) or getattr(original, "sqlstate", None)
    if sqlstate is None:
        cause = getattr(original, "__cause__", None)
        sqlstate = getattr(cause, "pgcode", None) or getattr(cause, "sqlstate", None)
    if not isinstance(sqlstate, str):
        return isinstance(exc, OperationalError)
    return sqlstate.startswith("08") or sqlstate in {
        "25006", "57P01", "57P02", "57P03"
    }


def run_migrations_offline() -> None:
    context.configure(
        url=sync_url.render_as_string(hide_password=False),
        target_metadata=target_metadata,
        include_schemas=True,
        version_table_schema=SCHEMA,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_once() -> None:
    engine = create_engine(
        sync_url,
        poolclass=pool.NullPool,
        echo=False,
        connect_args={"connect_timeout": 5},
    )
    try:
        with engine.begin() as connection:
            connection.exec_driver_sql(
                f'CREATE SCHEMA IF NOT EXISTS "{SCHEMA}"'
            )
            context.configure(
                connection=connection,
                target_metadata=target_metadata,
                include_schemas=True,
                version_table_schema=SCHEMA,
            )
            with context.begin_transaction():
                context.run_migrations()
    finally:
        engine.dispose()


def run_migrations_online() -> None:
    attempt = 0
    while True:
        attempt += 1
        try:
            run_migrations_once()
            logger.info("Alembic завершил миграции после попытки %d", attempt)
            return
        except DBAPIError as exc:
            if not is_temporary_database_error(exc):
                raise
            if attempt == 1 or attempt % 12 == 0:
                logger.warning(
                    "PostgreSQL пока недоступен для Alembic; "
                    "попытка %d, повтор через 5 секунд",
                    attempt,
                )
            time.sleep(5)


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

        validation_retry_seconds=settings.gigachat_validation_retry_seconds,
    )
    deposit_agent = PromptAnalysisAgent(
        prompt_loader,
        provider,
        block="deposit",
        log_responses=settings.log_llm_responses,
        validation_max_attempts=settings.gigachat_validation_max_attempts,
        validation_retry_seconds=settings.gigachat_validation_retry_seconds,
    )
    risk_agent = PromptAnalysisAgent(
        prompt_loader,
        provider,
        block="risk",
        log_responses=settings.log_llm_responses,
        validation_max_attempts=settings.gigachat_validation_max_attempts,
        validation_retry_seconds=settings.gigachat_validation_retry_seconds,
    )

    prompt_dialog_analyzer = PromptDialogAnalyzer(
        prompt_loader=prompt_loader,
        hygiene_agent=hygiene_agent,
        deposit_agent=deposit_agent,
        risk_agent=risk_agent,
        settings=settings,
    )

    external_analyzers = {}
    if settings.audio_analyzer_url:
        external_analyzers["audio"] = ExternalAnalyzerAgent(
            name="audio",
            base_url=settings.audio_analyzer_url,
            timeout_seconds=settings.external_analyzer_timeout_seconds,
        )
    if settings.factor_analyzer_url:
        external_analyzers["factor"] = ExternalAnalyzerAgent(
            name="factor",
            base_url=settings.factor_analyzer_url,
            timeout_seconds=settings.external_analyzer_timeout_seconds,
        )

    dialog_orchestrator = DialogAnalysisOrchestrator(
        repository=dialog_repository,
        prompt_analyzer=prompt_dialog_analyzer,
        external_analyzers=external_analyzers,
        settings=settings,
        positive_templates=prompt_loader.positive_templates,
    )

    worker = AnalysisWorker(
        settings=settings,
        tasks=task_repository,
        dialogs=dialog_repository,
        dialog_orchestrator=dialog_orchestrator,
    )

    dashboard_builder = DashboardBuilder(
        settings,
        dashboard_repository,
        dialog_report_repository,
    )

    scheduler = DashboardScheduler(
        settings,
        dashboard_builder,
        dashboard_repository,
        dialog_repository,
        task_repository,
    )

    app.state.analysis_worker = worker
    app.state.dashboard_scheduler = scheduler
    app.state.dialog_intake = DialogIntakeService(
        settings,
        dialog_repository,
    )
    app.state.manager_service = ManagerService(
        dashboard_repository,
        dialog_report_repository,
        settings,
    )
    app.state.dashboard_test_data = (
        DashboardTestDataService(
            settings,
            prompt_loader,
            dialog_report_repository,
            dashboard_repository,
            dashboard_builder,
        )
        if settings.test_data_api_enabled
        else None
    )

    worker_task = asyncio.create_task(worker.run(), name="analysis-worker")
    scheduler.start()

    try:
        yield
    finally:
        scheduler.shutdown()
        await worker.stop()
        worker_task.cancel()
        await asyncio.gather(worker_task, return_exceptions=True)
        close_operations = []
        for agent in external_analyzers.values():
            close_operations.append(agent.close())
        await asyncio.gather(*close_operations, return_exceptions=True)
        await provider.close()
        await get_database_engine().dispose()


app = FastAPI(title="AI-Q-SCORE", lifespan=lifespan)
app.include_router(callback_router, prefix=settings.app_api_prefix)
app.include_router(manager_router, prefix=settings.app_api_prefix)
app.include_router(runs_router, prefix=settings.app_api_prefix)
app.include_router(health_router, prefix=settings.app_api_prefix)
app.mount(settings.prometheus_path, make_asgi_app())
